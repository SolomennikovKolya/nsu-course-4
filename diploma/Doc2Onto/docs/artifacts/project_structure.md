# Структура проекта

```python
Doc2Onto/
├── README.md                          # Этот файл
├── main.py                            # Точка входа
├── requirements.txt                   # Зависимости
├── .env.example                       # Пример переменных окружения
├── docs/                              # Документы для диплома и вспомогательные файлы
│
├── app/                               # << Оркестрация приложения >>
│   ├── pipeline.py                    # Пайплайн обработки документов (5 стадий)
│   ├── context.py                     # Глобальный контекст (логгер, менеджеры, репозиторий онтологии)
│   ├── settings.py                    # Пути (.data, ресурсы), константы
│   ├── agents.py                      # Вызов LLM (OpenAI Responses API)
│   └── logger.py                      # Логгеры
│
├── models/                            # << DTO >>
│   ├── document.py                    # Документ + контекст для пайплайна (Document.Status — enum состояний)
│   ├── template.py                    # Шаблон + контекст для пайплайна + загрузчик кода шаблона
│   └── extraction_result.py           # Результат экстрактора (template/llm/normalized + ошибки по полям)
│
├── modules/                           # << Шаги пайплайна >>
│   ├── base.py                        # Базовый модуль + результат выполнения модуля
│   ├── converter/                     # Конвертер (со всеми специализированными конвертерами)
│   ├── classifier.py                  # Классификатор (выбор шаблона)
│   ├── extractor.py                   # Экстрактор (декларативное извлечение → LLM-коррекция/извлечение → нормализация)
│   ├── builder.py                     # GraphBuilder для построения чернового RDF-графа
│   ├── connector.py                   # Коннектор (применение правок, merge с онтологией)
│   └── reconciler.py                  # Reconciler - НЕ отдельный модуль (необязательное согласование IRI с уже существующими)
│
├── core/                              # << Ядро (всё оно используется в коде шаблона) >>
│   ├── template/                      # База шаблона
│   │   ├── base.py                    # Контракт кода шаблона (BaseTemplateCode)
│   │   └── validation.py              # Структурная валидация шаблона (AST-анализ, dry-run, сверка с онтологией)
│   ├── uddm/                          # Унифицированный формат представления документов
│   │   ├── model.py                   # Все элементы UDDM (Root, Text, P, ListBlock, Item, Table, Row, Cell)
│   │   ├── schema.xsd                 # XSD-схема UDDM
│   │   └── algorithms.py              # Дополнительные алгоритмы для работы с UDDM
│   ├── fields/                        # DSL для построения полей
│   │   ├── field.py                   # Определение поля (name, description, selector, extractor, normalizer)
│   │   ├── field_selector.py          # Селектор поля (поиск элемента UDDM)
│   │   ├── field_extractor.py         # Экстрактор поля (предварительное преобразование строкового представления UDDM элемента)
│   │   └── field_normalizer.py        # Нормализатор поля (приведение к канонической форме)
│   ├── concepts/                      # Концепты онтологии (identity-стратегии для классов и типов данных)
│   │   ├── base.py                    # BaseConcept, ConceptParts, ConceptKind
│   │   ├── person.py                  # Персона (морфологическое приведение ФИО)
│   │   ├── organization.py            # Организация
│   │   ├── group.py                   # Учебная группа
│   │   ├── direction.py               # Направление подготовки
│   │   ├── profile.py                 # Профиль направления
│   │   ├── thesis.py                  # ВКР (composite IRI от студента)
│   │   ├── practice.py                # Практика (composite IRI от студента и даты начала)
│   │   ├── position.py                # Должность (перечисление)
│   │   ├── degree.py                  # Учёная степень (перечисление)
│   │   ├── title.py                   # Учёное звание (перечисление)
│   │   ├── grade.py                   # Итоговая оценка (перечисление)
│   │   ├── date.py                    # Дата (xsd:date)
│   │   ├── email.py                   # Email
│   │   ├── telephone.py               # Телефон
│   │   ├── _hash.py                   # short_sha1 (хеширование IRI)
│   │   ├── _morph.py                  # Морфология ФИО (pymorphy3): род и именительный падеж
│   │   └── _enum.py                   # Базовая инфраструктура для концептов-перечислений
│   └── graph/                         # DSL для построения графа
│       ├── template_graph_builder.py  # Сборщик графа (TemplateGraphBuilder, ONTO, ValueProxy)
│       ├── draft_graph.py             # Черновое представление графа (DraftGraph) и редактирование (EditedGraph)
│       └── rdflib_draft_outer.py      # Обёртка для интеграции draft-графа с rdflib (XSD, RDF, OWL, RDFS)
│
├── storage/                           # << Работа с файлами >>
│   ├── base_manager.py                # Базовый менеджер хранилища
│   ├── document_manager.py            # Менеджер документов
│   ├── template_manager.py            # Менеджер шаблонов
│   └── ontology_repository.py         # Чтение TBox, журнал слияния, история документов, политики mergePolicy
│
├── utils/                             # << Утилиты вне ядра >>
│   ├── general.py                     # Парсинг словарных полей и т. п.
│   └── ontology_summary.py            # Текстовая сводка по онтологии (для UI и промптов)
│
├── scripts/                           # << Вспомогательные скрипты >>
│   └── gen_rdflib_draft_outer.py      # Генерация обёрток над rdflib
│
├── ui/                                # << GUI (PySide6) >>
│   ├── main_window.py                 # Главный класс и логика основного окна приложения
│   ├── common/                        # Переиспользуемые виджеты и оформление
│   │   ├── design.py                  # Палитры и стили
│   │   └── editable_title.py          # Редактируемый заголовок
│   ├── documents/                     # Вкладка для работы с документами
│   │   ├── documents_tab.py           # Вкладка со списком документов
│   │   ├── doc_info.py                # Виджет с информацией о документе
│   │   ├── doc_view.py                # Контейнер с тремя суб-вкладками просмотра документа
│   │   ├── status_bar.py              # Отображаемый статус документа
│   │   └── view/                      # Подмодули представления документа
│   │       ├── original_tab.py        # Предпросмотр оригинала документа (PDF + кнопка «открыть в системной программе»)
│   │       ├── uddm_tab.py            # Вкладка для просмотра UDDM (HTML и tree-view)
│   │       ├── graph_tab.py           # Двухпанельный редактор извлечённого графа фактов
│   │       └── common.py              # Общие компоненты и вспомогательные функции для view
│   ├── templates/                     # Вкладка для работы с шаблонами
│   │   ├── templates_tab.py           # Вкладка со списком шаблонов
│   │   ├── temp_info.py               # Виджет с информацией о шаблоне
│   │   └── python_code_html.py        # Подсветка кода шаблона в HTML
│   └── ontology/                      # Вкладка для работы с онтологией
│       └── ontology_tab.py            # Иерархия классов, карточки индивидов и классов, диалог «Источник факта»
│
├── resources/                         # << Статические ресурсы >>
│   ├── prompts/                       # Промпты для LLM
│   ├── template/
│   │   └── code_example.py            # Пустой пример кода шаблона (который LLM должен заполнить при генерации)
│   ├── onto/
│   │   └── schema.ttl                 # Схема онтологии (read-only TBox)
│   └── images/                        # Иконки и изображения для UI
│
└── .data/                             # << Динамические данные (создаётся при работе приложения) >>
    ├── documents/                     # Документы системы (по подкаталогу на каждый id)
    ├── templates/                     # Шаблоны системы (по подкаталогу на каждый id)
    ├── ontology/                      # Онтология
    │   ├── schema.ttl                 # Копия схемы онтологии для сборки
    │   ├── history.json               # Порядок добавления документов в модель (источник истины для ABox)
    │   ├── ontology.ttl               # Полная собранная модель (TBox + ABox)
    │   └── facts.jsonl                # Append-only журнал событий слияния (add / retract / reject)
    └── logs/                          # Логи
        ├── app.log                    # Лог работы приложения
        └── agents.log                 # Лог работы LLM-агентов (полные prompt + response)
```
